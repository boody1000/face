// دالة التبديل بين الصفحات عند الضغط على الأزرار function switchPage(pageId, event) { // 1. إخفاء جميع الصفحات const pages = document.querySelectorAll('.page'); pages.forEach(page => { page.classList.remove('active-page'); }); // 2. إزالة التحديد عن جميع أزرار التنقل const buttons = document.querySelectorAll('.nav-btn'); buttons.forEach(btn => { btn.classList.remove('active'); }); // 3. إظهار الصفحة المطلوبة فقط document.getElementById(pageId).classList.add('active-page'); // 4. تمييز الزر الذي تم النقر عليه if (event && event.currentTarget) { event.currentTarget.classList.add('active'); } }
